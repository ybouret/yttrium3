\section{Definitions}

\begin{itemize}

\item Let us define $\vec F(x,\vec a)$ a function from $\mathbb R$ to $\mathbb R^d$ with parameters $\vec a$.
\item Let us define $S_j$ a sample with $N_j$ points $(x_{ji},\vec y_{ji})$.
\item We define the squared distance:
\begin{equation}
	D^2_j = \frac{1}{2}\sum_{i=1}^{N_j} \left[\vec y_{ji} - \vec F(x_{ji},\vec a) \right]^2
\end{equation}

\item We define the total squared distance:
\begin{equation}
	 D^2 = \sum_{j=1}^M D^2_j
\end{equation}

\end{itemize}

We compute the gradient using:
\begin{equation}
	\partial_{a_k} D^2_j = - \sum_{i=1}^{N_j} \langle  \partial_{a_k} \vec F(x_{ji},\vec a) \vert \vec y_{ji} - \vec F(x_{ji},\vec a) \rangle
\end{equation}
We define:
\begin{equation}
	\beta_{jk} = -\partial_{a_k} D^2_j = \sum_{i=1}^{N_j} \langle  \partial_{a_k} \vec F(x_{ji},\vec a) \vert \vec y_{ji} - \vec F(x_{ji},\vec a) \rangle
\end{equation}
and:
\begin{equation}
	\beta_k = \sum_j \beta_{jk}
\end{equation}

We compute:
\begin{equation}
\partial_{a_l} \beta_{jk} =  \underbrace{\sum_{i=1}^{N_j} 
\langle \partial^2_{a_ka_l} \vec F(x_{ji},\vec a) \vert \vec y_{ji} - \vec F(x_{ji},\vec a)\rangle 
}_{\beta'_{jkl}}
- \underbrace{\sum_{i=1}^{N_j}\langle  \partial_{a_k} \vec F(x_{ji},\vec a) \vert  \partial_{a_l} \vec F(x_{ji},\vec a) \rangle}
_{\alpha_{jkl}}
\end{equation}

We define the matrix:
\begin{equation}
	\bm \alpha_j = 
	\begin{bmatrix}
	\cdots & \cdots & \cdots \\
	\cdots & \sum_{i=1}^{N_j}\langle  \partial_{a_k} \vec F(x_{ji},\vec a) \vert  \partial_{a_l} \vec F(x_{ji},\vec a) \rangle & \cdots\\
	\cdots & \cdots & \cdots \\
	\end{bmatrix}
\end{equation}
and likewise:
\begin{equation}
	\bm \beta'_j = 
	\begin{bmatrix}
	\cdots & \cdots & \cdots \\
	\cdots & \sum_{i=1}^{N_j} \langle \partial^2_{a_ka_l} \vec F(x_{ji},\vec a) \vert \vec y_{ji} - \vec F(x_{ji},\vec a)\rangle  & \cdots\\
	\cdots & \cdots & \cdots \\
	\end{bmatrix}
\end{equation}

We want:
\begin{equation}
	\vec \beta(\vec a) = \vec 0
\end{equation}
