\section{Approximation: Michaelis-Menten}

\begin{equation}
 S + E \xrightleftharpoons[k_d]{k_f} ES \xrightarrow[]{k_p} E + P,\;\; E_0 = [E] + [ES]
\end{equation}

\begin{equation}
\left\lbrace
\begin{array}{ccl}
\partial_t [S]  & = & v_d - v_f\\
\partial_t [P]  & = & v_p \\
\partial_t [ES] & = & v_f - v_d - v_p\\
v_d & = & k_d [ES]\\ 
v_p & = & k_p [ES]\\
v_f & = & k_f [E][S]\\
\end{array}
\right.
\end{equation}

Using:
\begin{equation}
k_b = k_d + k_p
\end{equation}

\begin{equation}
	[ES] \simeq E_0 \dfrac{[S]}{[S]+\dfrac{k_b}{k_f}}
\end{equation}

\begin{equation}
	\partial_t [P] = -\partial_t [S] \simeq k_p E_0 \dfrac{[S]}{[S]+\dfrac{k_b}{k_f}}
\end{equation}
