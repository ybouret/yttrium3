\begin{itemize}
\item We assume that we have a local minimum bracketing such that $a\leq b \leq c$ and $f_b \leq f_a$ and $f_b \leq f_c$.
\item We transform $a$ into $0$, $c$ into $1$ and $b$ into $\beta$.
\item We transform $f_a-f_b$ into $\alpha \geq 0$ a and $f_c-f_b$ into $\gamma \geq 0$.
\item The reduced equation is:
\begin{equation}
	g(u) = \alpha + (\gamma-\alpha) u - \kappa u(1-u)
\end{equation}

\item $\kappa$ is defined by $g(\beta) = 0$:
\begin{equation}
	\kappa = \dfrac{(1-\beta)\alpha + \beta \gamma }{(1-\beta)\beta}
\end{equation}

\item By differentiation, the expected minimum is at:
\begin{equation}
	u_m = \dfrac{1}{2} \left\lbrack 1 - \beta(1-\beta) \dfrac{\gamma-\alpha}{\beta \gamma + (1-\beta)\alpha } \right\rbrack
\end{equation}

\item The special case are:
	\begin{itemize}
	\item
		\begin{equation}
			\lim_{\alpha\simeq\gamma} = \dfrac{1}{2}
		\end{equation}
	\item 
		\begin{equation}
			\forall (\alpha,\gamma) \; \lim_{\beta\to0} u_m = \lim_{\beta\to1} u_m = \dfrac{1}{2}
		\end{equation}
		
	\item
		\begin{equation}
			\forall \gamma>0, \; \lim_{\alpha\to0} u_m = \dfrac{\beta}{2}
		\end{equation}
	\item
		\begin{equation}
			\forall \alpha>0, \; \lim_{\gamma\to0} u_m = \dfrac{1+\beta}{2}
		\end{equation}
	\end{itemize}

	
\end{itemize}
